{
  printf '\n===== START: SCRIPT2 1.0.11 RELEASE =====\n'
  if bash <<'SCRIPT2_RELEASE'
script2_update_and_publish() {
  cd "$HOME/devProjects/homebridge-script2" || return
  script2_changes="$(git status --porcelain)" || return
  if [ -n "$script2_changes" ]; then
    printf '%s\n' 'STOP: local changes found. Commit or move them before updating.'
    return 1
  fi
  script2_remote="$(git remote get-url origin)" || return
  case "$script2_remote" in
    git@github.com:pponce/homebridge-script2.git|https://github.com/pponce/homebridge-script2.git|https://github.com/pponce/homebridge-script2|ssh://git@github.com/pponce/homebridge-script2.git) ;;
    *) printf '%s\n' 'STOP: unexpected origin repository.'; return 1 ;;
  esac
  git remote set-url origin git@github.com:pponce/homebridge-script2.git || return
  git fetch origin master || return
  if git show-ref --verify --quiet refs/heads/master; then
    git switch master || return
  else
    git switch --track -c master origin/master || return
  fi
  git merge --ff-only origin/master || return
  script2_local="$(git rev-parse HEAD)" || return
  script2_upstream="$(git rev-parse origin/master)" || return
  if [ "$script2_local" != "$script2_upstream" ]; then
    printf '%s\n' 'STOP: local master has unpublished commits.'
    return 1
  fi
  node -e 'if(require("./package.json").version!=="1.0.11")throw Error("Expected Script2 1.0.11.");' || return
  bash scripts/publish-stable.sh || return
  git fetch origin --tags || return
}
script2_update_and_publish
SCRIPT2_RELEASE
  then
    printf '\n===== END: SCRIPT2 1.0.11 RELEASE — SUCCESS =====\n'
  else
    script2_release_status=$?
    printf '\n===== END: SCRIPT2 1.0.11 RELEASE — FAILED (status %s) =====\n' "$script2_release_status"
  fi
}
